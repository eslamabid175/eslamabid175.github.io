dart enum AuthStatus { initial, loading, authenticated, unauthenticated, error } class AuthState extends Equatable { final AuthStatus status; final AuthUser? user; final String? errorMessage; const AuthState({ this.status = AuthStatus.initial, this.user, this.errorMessage, }); AuthState copyWith({AuthStatus? status, AuthUser? user, String? errorMessage}) { return AuthState( status: status ?? this.status, user: user ?? this.user, errorMessage: errorMessage, ); } } class AuthCubit extends Cubit<AuthState> { final AuthRemoteDataSource _dataSource; final SecureStorageService _storage; AuthCubit({required AuthRemoteDataSource dataSource, required SecureStorageService storage}) : _dataSource = dataSource, _storage = storage, super(const AuthState()); Future<void> checkAuth() async { final token = await _storage.getAccessToken(); if (token == null) { emit(const AuthState(status: AuthStatus.unauthenticated)); return; } try { final user = await _dataSource.getProfile(); emit(AuthState(status: AuthStatus.authenticated, user: user)); } catch (_) { await _storage.clearTokens(); emit(const AuthState(status: AuthStatus.unauthenticated)); } } }

Performance Best Practices

🔧
Use const Widgets

Isolates

Dispose Correctly

🖼️
Image Optimization
استخدم lazy loading + caching + أحجام صور مناسبة لتقليل استهلاك الذاكرة وتسريع الـ rendering.

Architecture Deep Dive

💡 المعمارية الكويسة هدفها حاجتين: إن التغيير في مكان ما يكسرش مكان تاني، وإن كل حتة أقدر أختبرها لوحدها. ده اللي ثبّته في دماغي درس المعمارية في الـ Senior / Lead Roadmap.
🧩
Presentation
الشاشات والـ state management (Cubit / Notifier). ما تعرفش حاجة عن HTTP ولا JSON.
🏛️
Domain
Entities (classes Dart عادية) + use cases + الـ interfaces بتاعة الـ repositories. منطق صافي من غير Flutter ولا أي package.
🗄️
Data
تنفيذ الـ repositories + مصادر البيانات (API عن طريق dio، وتخزين محلي زي drift) + الـ DTOs اللي فيها fromJson.
🔁
قاعدة الاعتمادية
الـ Presentation والـ Data الاتنين بيعتمدوا على الـ Domain، والـ Domain ما بيعتمدش على حد. الـ Data بتنفّذ interface الـ Domain عرّفه = Dependency Inversion.

Domain بيعرّف «إيه»، Data بتنفّذ «إزاي»

dart // domain: بيعرّف إيه المطلوب abstract class CourseRepository { Future<List<Course>> getCourses(); } // data: بتنفّذ إزاي class CourseRepositoryImpl implements CourseRepository { CourseRepositoryImpl(this._api, this._local); final CourseApi _api; final CourseLocalStore _local; @override Future<List<Course>> getCourses() async { final dtos = await _api.fetchCourses(); final courses = dtos.map((d) => d.toEntity()).toList(); await _local.saveAll(courses); return courses; } }

Bloc ولا Riverpod؟

💡 السؤال الحقيقي مش «مين أحسن»: السؤال: الحالة عايشة فين؟ ومين يقدر يغيّرها؟ والشاشة بتتحدث إمتى؟
🧱
Bloc / Cubit
events بتدخل و states بتطلع. كود متكرر أكتر، بس تتبّع التغييرات واضح جدًا (كل تغيير ليه event). الـ DI بـ RepositoryProvider أو get_it، والـ loading/error بكتبهم بنفسي. مناسب لـفرق كبيرة وflows معقدة.
🌿
Riverpod
providers بتوفّر قيم وحالة. كود أقل، الـ DI مدمج، وAsyncValue جاهز للـ async. أقل صرامة في التتبّع. مناسب لـفرق صغيرة ومتوسطة وسرعة في التطوير.
⚖️ الـ Trade-off: الـ 3 طبقات في MVP صغير أو تطبيق عمره قصير بتضيف كود كتير من غير فايدة، والـ use case اللي كل شغلته إنه ينادي الـ repository ممكن يتشال. الحكم ده نفسه جزء من كونك سنيور: كبّر المعمارية على قد حجم الفريق وعمر التطبيق. واللي بيبوّظ المشاريع مش اختيار Bloc أو Riverpod، هو إن كل شاشة تتكتب بطريقة، أو إن الـ business logic تتسرب جوه الـ widgets.
🧪 اختبر نفسك — جاوب في دماغك الأول، وبعدين دوس على السؤال
ليه طبقة الـ Domain ما ينفعش تعمل import لـ dio أو Flutter؟

لأنها لو اعتمدت عليهم، أي تغيير في المكتبة أو في الشاشات هيوصل للمنطق، ومش هقدر أختبرها بسرعة في Dart عادي. الـ Domain بيعرّف الـ interfaces، والطبقات التانية هي اللي تتكيف معاه.

فريق 8 مطورين وتطبيق بنكي: Bloc ولا Riverpod؟

غالبًا Bloc: كل تغيير في الحالة ليه event واضح وسهل تتبعه وتسجله، ودا مهم في تطبيق مالي وفريق كبير محتاج قواعد صارمة. Riverpod ينفع برضه لو الفريق متفق على conventions واضحة؛ المهم إن الكل يمشي بنفس الطريقة.

Offline-first & Sync

💡 الفكرة في جملة: في التطبيق الـ offline-first، الداتابيز المحلية هي مصدر الحقيقة للشاشات، والسيرفر بيتزامن معاها في الخلفية. الشاشة عمرها ما تقرا من الـ API مباشرة؛ بتسمع على stream من الداتابيز المحلية، فوجود النت أو غيابه ما بيغيّرش كود الشاشة.
UI (Flutter)
←
Local DB (drift)
←
Outbox
←
Sync engine
←
Server

الكتابة: الـ Outbox Pattern

1️⃣
اكتب محلي فورًا
التعديل يتكتب في الداتابيز المحلية على طول، فالشاشة تتحدث فورًا (optimistic UI).
2️⃣
نفس الـ transaction
في نفس الـ transaction ضيف العملية في جدول outbox: طابور العمليات اللي لسه ما اتبعتتش.
3️⃣
Sync engine
بيبعت العمليات بالترتيب لما النت يرجع، ويعيد المحاولة بـ backoff لو فشلت.
4️⃣
السيرفر أكّد؟
امسح العملية من الـ outbox. لو وصلت مرتين، السيرفر بيتجاهل المكرر بالـ id.
dart class PendingOps extends Table { TextColumn get id => text()(); // UUID من الموبايل = idempotency key TextColumn get entity => text()(); // 'sale' TextColumn get operation => text()(); // 'create' | 'update' | 'delete' TextColumn get payload => text()(); // JSON IntColumn get attempts => integer().withDefault(const Constant(0))(); DateTimeColumn get createdAt => dateTime()(); @override Set<Column> get primaryKey => {id}; }
dart // الكتابة: الصف + العملية في الـ outbox في نفس الـ transaction Future<void> addSale(Sale sale) { return _db.transaction(() async { await _db.into(_db.sales).insert(sale.toCompanion()); await _db.into(_db.pendingOps).insert(PendingOpsCompanion.insert( id: _uuid.v4(), // idempotency key للعملية entity: 'sale', operation: 'create', payload: jsonEncode(sale.toJson()), createdAt: DateTime.now(), )); }); } // القراية: الشاشة بتسمع على stream من القاعدة المحلية، مش من الـ API Stream<List<Sale>> watchSales() { return _db.select(_db.sales).watch().map( (rows) => rows.map((r) => r.toEntity()).toList(), ); }
dart Future<void> flush() async { final ops = await (_db.select(_db.pendingOps) ..orderBy([(o) => OrderingTerm(expression: o.createdAt)])) .get(); for (final op in ops) { try { await _remote.push(op); // السيرفر بيتجاهل الـ id المكرر await (_db.delete(_db.pendingOps)..where((o) => o.id.equals(op.id))).go(); } catch (_) { await (_db.update(_db.pendingOps)..where((o) => o.id.equals(op.id))) .write(PendingOpsCompanion(attempts: Value(op.attempts + 1))); _scheduleRetry(Duration(seconds: 1 << min(op.attempts, 8))); // backoff break; // نحافظ على الترتيب: ما نعديش العملية اللي فشلت } } }
💡 ليه الـ UUID بيتولد على الموبايل؟ السجل ليه id ثابت من أول لحظة حتى لو اتعمل offline، فأقدر أربط بيه حاجات تانية، والسيرفر يقدر يتجاهل العملية لو وصلته مرتين. دي نفس فكرة الـ idempotency key في درس تصميم الـ API.

القراية: Delta sync

🕒
updated_at / version
على كل جدول بيتزامن، عشان أسأل السيرفر: «إيه اللي اتغير من آخر مزامنة؟» بدل ما أحمّل كل البيانات كل مرة.
🗑️
Soft delete
عمود deleted_at بدل المسح الفعلي. لو الصف اتمسح من الداتابيز، مفيش حاجة أقول بيها للموبايل إنه اتمسح.
🔗
Changes endpoint
زي GET /api/changes?since=<cursor> بيرجّع اللي اتعدّل واللي اتمسح من بعد الـ cursor.

الـ Conflicts: جهازين عدّلوا نفس الحاجة offline، مين يكسب؟

⏱️
Last write wins
الأحدث يكسب. بسيط، بس ممكن يضيّع بيانات، وساعة الموبايل مش مضمونة، فاعتمد على version من السيرفر.
📏
Server / Client wins
قاعدة ثابتة حسب نوع البيانات.
🧬
Field-level merge
لو كل جهاز عدّل field مختلف، خد الاتنين.
🙋
اسأل المستخدم
للبيانات المهمة، زي الـ conflicts في Git.

مش كل تطبيق محتاج كل ده

1
Cache للقراية
يعرض آخر بيانات اتحملت لما النت يقطع. التعقيد: قليل. (زي تطبيق أخبار)
2
قراية offline كاملة
داتابيز محلية + delta sync. التعقيد: متوسط.
3
كتابة offline
outbox + حل الـ conflicts. التعقيد: عالي جدًا.
🛠️ من شغلي الحقيقي — Mksaby: Mksaby تطبيق offline-first من المستوى التالت: Drift (SQLite) هو مصدر الحقيقة بـ 22 جدول بيتزامنوا، والتعديلات بتتسجل محليًا وبتترفع لـ Cloud Firestore عن طريق offline outbox sync. والموظفين بيدخلوا من موبايلاتهم وبيتزامنوا live مع صاحب المحل. يعني البيع والمشتريات والمخزون شغالين حتى والنت قاطع.
⚖️ الـ Trade-off: كل مستوى أعلى = تعقيد أكتر بكتير (outbox، retries، conflicts، soft delete على الـ backend). اختار المستوى على قد احتياج المستخدم الحقيقي، وقرّر بدري: التحويل للمستوى 3 بعد ما التطبيق يكبر مكلّف جدًا. تطبيق أخبار كفاية عليه المستوى 1، لكن حد بيسجّل عمليات في مكان النت فيه ضعيف محتاج المستوى 3.
🧪 اختبر نفسك — جاوب في دماغك الأول، وبعدين دوس على السؤال
ليه الـ soft delete ضروري في الـ delta sync؟

لأن الـ delta sync بيسأل «إيه اللي اتغير من آخر مرة»، والصف اللي اتمسح فعليًا مش هيرجع في أي نتيجة. الـ deleted_at بيخلي المسح نفسه تغيير يتبعت للموبايل فيمسح نسخته.

ليه الـ id بيتولد على الموبايل مش على السيرفر؟

لأن السجل اللي اتعمل offline محتاج id من أول لحظة عشان أربط بيه حاجات تانية. وكمان السيرفر بيستخدمه كـ idempotency key، فلو العملية وصلت مرتين يعرف إنها نفس العملية.

عشان المراجع يعرف إن الـ refactoring المفروض ما يغيّرش أي سلوك، فيراجعه بعين، والتغيير الحقيقي بعين تانية. لو اتخلطوا، الـ bug بيستخبى وسط مئات السطور اللي اتنقلت.

5. الـ Code review ومراجعة كود الـ AI

💡 الـ review آخر فرصة: أمسك bug أو ثغرة قبل ما توصل للمستخدمين — مش بحث عن فواصل ومسافات. براجع بالترتيب ده من الأهم للأقل:
  • هل الكود صح؟ والـ edge cases متغطية؟
  • الأمان: الصلاحيات والـ injection.
  • البيانات: الـ migrations، والـ N+1، والـ transactions.
  • التصميم وسهولة القراية.
  • الـ tests موجودة؟
  • الأسماء والـ style — وده سيبه لأدوات زي dart format و ruff.
💬
أسلوب التعليق
بلطف ووضّح «ليه»، وفرّق بين اللي لازم يتصلح والملاحظة الصغيرة (ابدأها بـ nit:).
🤖
كود الـ AI
راجعه زي كود junior شاطر: شكله مقنع، بس ممكن يستخدم API مش موجود، أو ينسى edge case، أو يخالف طريقة المشروع. السطر اللي مش فاهمه ما يتعملوش merge.
🧪 اختبر نفسك — جاوب في دماغك الأول، وبعدين دوس على السؤال
إيه أكتر حاجة أركز عليها وانا براجع كود كتبه الـ AI؟

إنه ممكن يبان صح وهو غلط: API مش موجود، أو edge case منسي، أو ثغرة زي query مش متفلتر بالمستخدم. وأهم قاعدة: السطر اللي مش فاهمه ما يتعملوش merge.

Lead Skills

1. تقسيم الشغل والتقدير

🍰
Vertical slices
قسّم الـ feature لمهام كل واحدة يوم أو اتنين بالكتير، واقطعها شريحة رفيعة من الأول للآخر (الداتابيز + الـ API + شاشة Flutter) بدل ما أخلص الـ backend كله الأول. مشاكل الربط تظهر في الأسبوع الأول مش الأخير.
📏
التقدير مدى مش وعد
3 أرقام: متفائل O، متوقع M، متشائم P. ولو الفرق بين O و P كبير = فيه مجهول كبير ← اعمل spike (وقت محدد للاستكشاف) قبل ما تلتزم.
📣
قول بدري
التقدير بيشمل الـ tests والمراجعة والـ deploy مش الكود بس. والخبر الوحش بدري ممكن يتدار، ومتأخر بيبقى أزمة.
🧮 المعادلة: E = (O + 4M + P) ÷ 6 — مثال (بالأيام): O = 2, M = 4, P = 10 ← E = 28 ÷ 6 ≈ 4.7 يوم.
🧪 اختبر نفسك — جاوب في دماغك الأول، وبعدين دوس على السؤال
قدّرت مهمة بالأيام: O = 1, M = 3, P = 11. احسب E، والأرقام دي بتقولي إيه؟

E = (1 + 12 + 11) ÷ 6 = 4 أيام. والفرق الكبير بين 1 و 11 معناه إن فيه مجهول كبير، فالأحسن أعمل spike قبل ما ألتزم بميعاد.

مديري عايز feature في أسبوع وانا شايفها 3 أسابيع. أقوله إيه؟

أشرح ليه 3 أسابيع، وأعرض اختيارات: نسخة أصغر تطلع في أسبوع وفيها كذا وناقصها كذا، أو الـ feature كاملة في 3 أسابيع، أو نشيل حاجة تانية من الخطة. يختار وهو فاهم التمن.

2. الـ ADRs: اكتب ليه اخترت

💡 ADR = Architecture Decision Record: مستند قصير لكل قرار تقني مهم، متخزّن جوه الـ repo. بعد 6 شهور محدش هيفتكر ليه اخترنا Riverpod أو cursor pagination. بيمنع الفريق يعيد نفس النقاش، وبيفهّم أي حد جديد بسرعة.
md # ADR-0007: Cursor pagination للـ feed ## الحالة مقبول ## السياق الـ feed بيتحدث باستمرار، والـ offset pagination بيكرر عناصر في الـ infinite scroll. ## القرار هنستخدم CursorPagination مترتبة بـ -created_at. ## البدائل اللي اترفضت - Offset pagination: أبسط، بس بيكرر عناصر وبيبطأ في الصفحات البعيدة. ## العواقب - (+) الـ scroll ثابت وسريع. - (−) مفيش «روح لصفحة 10».
⚖️ الـ Trade-off: أهم جزء في الـ ADR هو «البدائل» و«العواقب» لأنهم بيجبروني أفكر في الـ trade-offs. وما أكتبش ADR لكل تفصيلة صغيرة؛ بس للقرارات الصعب الرجوع فيها أو اللي هتأثر على الفريق فترة طويلة (state management، شكل الـ pagination، طريقة الـ auth).
🧪 اختبر نفسك — جاوب في دماغك الأول، وبعدين دوس على السؤال
إيه أهم جزء في الـ ADR، وليه؟

البدائل اللي اترفضت والعواقب. هما اللي بيشرحوا ليه القرار اتاخد وإيه تمنه، فأي حد جديد يفهم من غير ما يعيد نفس النقاش.

3. الـ Incidents: لما production يقع

🧯
رجّع الخدمة الأول
rollback، أو اقفل الـ feature بـ feature flag، أو زوّد السيرفرات. فهم السبب ييجي بعدين وبهدوء.
🧭
شخص واحد بيقود
بيوزع والباقي بينفذوا. من غير كده 5 أشخاص بيجربوا 5 حاجات في نفس الوقت ويلخبطوا بعض.
📢
طمّن الناس بانتظام
«شغالين عليها، والتحديث الجاي بعد 30 دقيقة» — حتى لو مفيش جديد. وسجّل timeline.
📝
Postmortem من غير لوم
إيه اللي حصل، أثره، السبب الجذري (اسأل «ليه؟» كذا مرة)، إيه اللي نفع، ومهام لكل واحدة مسؤول.
💡 السؤال الصح: مش «مين غلط؟» — السؤال: «إيه في النظام سمح إن الغلطة دي توصل production؟» لأن الناس لو اتلاموا هيخبّوا غلطاتهم والمشكلة هتتكرر.
🧪 اختبر نفسك — جاوب في دماغك الأول، وبعدين دوس على السؤال
السيرفر وقع الساعة 2 بالليل. إيه أول 3 حاجات أعملها؟

أرجّع الخدمة بأسرع طريقة (rollback أو إيقاف الـ feature)، وأحدد مين بيقود ويوزع الشغل، وأطمّن الناس المتأثرين بتحديثات منتظمة. البحث عن السبب الجذري بعد ما الخدمة ترجع.

4. الكلام مع ناس مش تقنيين

💡 ترجم التقني لأثر على البيزنس: بدل «محتاجين نعمل refactor للـ auth module» ← «أي تعديل في تسجيل الدخول بياخد 3 أضعاف وقته. أسبوعين تنضيف هيقللوا وقت التعديلات الجاية للنص، ويقللوا المخاطر الأمنية.»
  • اعرض اختيارات، لكل واحد وقته وتكلفته ومخاطره، ومعاهم ترشيحك.
  • ابدأ بالخلاصة، والتفاصيل لو اتسألت.
  • «مش عارف لسه، هرد عليك الخميس» أحسن ألف مرة من رقم بتخمّنه.
🧪 اختبر نفسك — جاوب في دماغك الأول، وبعدين دوس على السؤال
حوّل الجملة دي لكلام بيزنس: «الداتابيز محتاجة indexes».

مثلًا: «صفحة التقارير بتاخد 10 ثواني والمستخدمين بيشتكوا. يومين شغل هيخلوها أقل من ثانية.» الأثر والتكلفة والنتيجة، من غير مصطلحات.

5. تعليم الـ juniors

💡 شغلتي: إن الـ junior يبقى قادر يحل المشكلة الجاية لوحده، مش إني أحلها انا أسرع.
  • ما آخدش منهم الكيبورد. أسأل أسئلة توجّههم وأسيبهم يوصلوا.
  • وقت للزنقة: اللي واقف في مشكلة أكتر من ساعتين يسأل، ما يفضلش يومين لوحده.
  • أزوّد الصعوبة بالتدريج: مهمة أصعب شوية كل مرة، مع دعم.
  • الـ code review فرصة تعليم: أشرح «ليه»، مش بس «غيّر ده».
  • السؤال والغلط لازم يبقوا آمنين: الفريق اللي بيخاف يسأل بيعمل غلطات أكبر.
  • في زمن الـ AI: أعلّمهم يفهموا الكود اللي الـ AI بيكتبه، مش بس يعملوله paste.
⚖️ الـ Trade-off: إني أصلّح المشكلة بنفسي أسرع النهارده، بس بيخلّي الفريق معتمد عليّا بكرة. التعليم أبطأ في الأول وأوفر بكتير بعد كده.
🧪 اختبر نفسك — جاوب في دماغك الأول، وبعدين دوس على السؤال
junior واقف في bug بقاله ساعة. أعمل إيه؟

ما أصلّحهوش بنفسي. أسأله جرّب إيه ووصل لحد فين، وأوجّهه بسؤال يقرّبه من المكان الصح. ولو واقف فعلًا، أديله خطوة واحدة يقف عليها وأسيبه يكمل.
